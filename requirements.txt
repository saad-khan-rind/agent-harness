# The IANA timezone database as a Python package. python:3.12-slim ships
# without /usr/share/zoneinfo, so zoneinfo falls back to this; without it
# every timezone lookup in mcp_server_time_plus fails inside the image
# while passing on a developer's machine (ADR-044).
tzdata==2026.2

# Web research server (mcp_server_web_search).
#
# ddgs: a metasearch library that aggregates several public search engines.
# No API key and no account, which is what lets the whole team run the agent
# and lets CI stay secret-free. It rotates across engines, so one engine
# throttling does not fail the call (ADR-055).
ddgs==9.16.0
# httpx already arrives with openai and mcp, but mcp_server_web_search now
# imports it directly. An undeclared transitive dependency is one that can
# vanish under an unrelated version bump.
httpx==0.28.1

# Vector database client for long-term memory (ADR-065).
#
# The dependency also carries an in-process "local mode"
# (`QdrantClient(":memory:")`), which is what lets the whole memory test
# suite run the real store against the real query engine with no server
# and no Docker — `.gitlab-ci.yml` has neither.
qdrant-client==1.15.1
# Transitive of qdrant-client, pinned because it is not. From 2.2 the
# default Linux wheel requires the x86-64-v2 ISA (SSE4.2 and friends).
# The FIM GitLab runner's VM does not expose those flags, so an unpinned
# install (2.5.x today) raises on the first `import qdrant_client` and
# every later test dies with "cannot load module more than once per
# process" (ADR-073). 2.1.3 is the last 2.x that still ships a v1 wheel.
# A developer laptop and this pin agree; CI and an unpinned install do not.
numpy==2.1.3

# Calendar server (mcp_server_calendar), a fork of ics-calendar-mcp.
#
# Both are upstream's dependencies, kept because the fork keeps upstream's
# behaviour (ADR-071).
#
# icalendar: reads and writes RFC 5545. BSD licensed.
# recurring-ical-events: expands RRULEs, so one "weekly standup" becomes an
# occurrence on every Monday. This is the capability that made forking
# worth more than writing our own — doing it by hand means implementing
# BYDAY, BYSETPOS, UNTIL, COUNT and EXDATE, which is a project.
#
# Not kept: httpx (upstream fetches a remote feed; we read a local file)
# and pytz (this repository uses zoneinfo and tzdata everywhere else).
icalendar==6.1.0
recurring-ical-events==3.8.2

# Metrics exposition for Week 3 observability (§8.4). Pinned like everything
# else here: the exposition format and the `start_http_server` return shape
# are both things we assert on, so a floating version could break a dashboard
# without a line of our code changing (ADR-081).
prometheus-client==0.26.0

# Optional OpenTelemetry export (ADR-091). Installed so the feature can be
# demonstrated without a second pip install, but imported lazily and only
# when `otlp_enabled` is true: the harness starts, runs and passes its whole
# test suite with these absent, which is the property that keeps the span
# model ours rather than the SDK's.
#
# Pinned together. The exporter and the SDK share an internal protobuf
# encoding that is not part of either one's public API, so a floating pair
# is a pair that will eventually disagree.
opentelemetry-sdk==1.40.0
opentelemetry-exporter-otlp-proto-http==1.40.0

# Local embedding model, for the deployment case this project actually has:
# InnKube serves chat but not /embeddings, so the memory subsystem fell back
# to a lexical hashing embedder that measures word overlap rather than
# meaning. "what do i like?" against a stored "Likes: swimming, ..." scored
# 0.115 against a floor of 0.20 and was never retrieved (ADR-066).
#
# fastembed rather than sentence-transformers, and the reason is the image.
# Both load the same MiniLM checkpoint and produce the same vectors;
# sentence-transformers reaches it through torch, which with scipy,
# scikit-learn and transformers is ~1.4 GB of wheels on a base image of
# ~150 MB. fastembed reaches it through onnxruntime in ~24 MB. It is also
# the Qdrant project's own library, so it is the closest thing to a
# supported pairing with the client pinned above.
#
# The model itself is fetched at build time (see Dockerfile), so a container
# runs with no network and no first-run download.
fastembed==0.8.0
